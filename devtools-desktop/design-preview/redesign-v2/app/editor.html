#!/bin/bash
# LDTS 后端本地开发启动脚本
set -euo pipefail
OFFLINE="false"
if [[ "${1:-}" == "--offline" ]]; then
OFFLINE="true"
echo "[offline] skip remote tunnel check"
fi
if [[ "$OFFLINE" != "true" ]]; then
./ldts-tunnel-daemon.sh status || ./ldts-tunnel-daemon.sh restart
fi
echo "backend listening on 127.0.0.1:13900"
exec java -jar target/app.jar --spring.profiles.active=dev
